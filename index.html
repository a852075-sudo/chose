⚙️ Google Apps Script 後端代碼與 Step-by-Step 部署說明

PWA 部署、試算表連結與 Google Apps Script 程式碼

請複製以下完整程式碼貼到 GAS (code.gs)： 
/**
 * 2026 尾牙投票系統 Google Apps Script 後端 (code.gs)
 * 支援功能：
 * 1. doGet: 回傳目前試算表的投票統計與同仁清單 (JSON)
 * 2. doPost: 接收實名制投票 (人員名稱, 選項, 備註, 時間戳記)
 * 3. 密碼重置: 驗證密碼 "tung" 後清空資料列 (保留第 1 列標題)
 */

function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : "read";
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("工作表1") || ss.getSheets()[0];
  
  if (action === "read") {
    var data = sheet.getDataRange().getValues();
    var rows = [];
    if (data.length > 1) {
      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        if (row[0]) {
          rows.push({
            timestamp: row[0],
            name: row[1],
            option: row[2],
            comment: row[3] || ""
          });
        }
      }
    }
    return ContentService
      .createTextOutput(JSON.stringify({ status: "success", data: rows }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  return ContentService
    .createTextOutput(JSON.stringify({ status: "ok" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("工作表1") || ss.getSheets()[0];
    
    // 初始化第一列標題 (若試算表為空)
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["時間戳記", "人員名稱", "投票選項", "備註"]);
    }
    
    var body = {};
    if (e.postData && e.postData.contents) {
      body = JSON.parse(e.postData.contents);
    } else if (e.parameter) {
      body = e.parameter;
    }
    
    // 處理重置功能 (密碼 "tung")
    if (body.action === "reset") {
      if (body.password === "tung") {
        var lastRow = sheet.getLastRow();
        if (lastRow > 1) {
          sheet.deleteRows(2, lastRow - 1);
        }
        return ContentService
          .createTextOutput(JSON.stringify({ status: "success", message: "投票紀錄已成功清空！" }))
          .setMimeType(ContentService.MimeType.JSON);
      } else {
        return ContentService
          .createTextOutput(JSON.stringify({ status: "error", message: "密碼驗證失敗，無法重置！" }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
    
    // 處理投票寫入
    var name = body.name || "";
    var option = body.option || "";
    var comment = body.comment || "";
    var timestamp = new Date();
    
    if (!name || !option) {
      return ContentService
        .createTextOutput(JSON.stringify({ status: "error", message: "缺少必要欄位！" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // 檢查實名制是否已有紀錄 (一人一票，若有則覆寫或更新)
    var data = sheet.getDataRange().getValues();
    var existingRowIndex = -1;
    for (var r = 1; r < data.length; r++) {
      if (data[r][1] === name) {
        existingRowIndex = r + 1; // 1-based index
        break;
      }
    }
    
    if (existingRowIndex > 0) {
      // 覆寫更新
      sheet.getRange(existingRowIndex, 1, 1, 4).setValues([[timestamp, name, option, comment]]);
    } else {
      // 新增一行
      sheet.appendRow([timestamp, name, option, comment]);
    }
    
    return ContentService
      .createTextOutput(JSON.stringify({ status: "success", message: "投票已成功記錄！" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

📋 完整 Google Sheet + GAS Web App 部署步驟：

  1. 建立試算表： 開啟 Google Drive 建立新的「Google 試算表」，命名為「2026尾牙投票統計」。
  2. 設定表頭： 在第 1 列依序輸入：時間戳記 | 人員名稱 | 投票選項 | 備註。
  3. 發佈到網路 (CSV 唯讀)： 點擊試算表選單【檔案】➔【共用】➔【發佈到網路】➔ 選擇「整份文件」或「工作表1」，格式選擇「以逗號分隔的值 (.csv)」，點擊【發佈】。（您已提供的 CSV 網址即是此方式產生）。
  4. 建立 Apps Script： 在試算表上方選單點擊【擴充功能】➔【Apps Script】。
  5. 貼上代碼： 將左側頁籤的 code.gs 程式碼完整覆蓋貼入編輯器中，並按 Ctrl+S (Cmd+S) 儲存。
  6. 部署為網頁應用程式 (重要！)：
    • 點擊右上角藍色【部署】➔【新部署作業】。
    • 齒輪圖示選擇【網頁應用程式 (Web App)】。
    • 說明輸入：「2026尾牙投票 API」。
    • 執行身分：選擇「我 (your_email@gmail.com)」。
    • 誰可以存取：務必選擇「所有人 (Anyone)」（這樣同仁在手機投票才不需登入 Google）。
    • 點擊【部署】，若跳出授權視窗，點擊「審查權限」➔ 選擇您的帳號 ➔ 點擊「進階 (Advanced)」➔ 點擊「前往專案 (不安全)」➔ 點擊「允許」。
  7. 複製 Web App 網址： 複製產生的「網頁應用程式網址」 (以 https://script.google.com/macros/s/.../exec 結尾)，貼到本系統設定中的「GAS API 網址」欄位即可無縫對接！

sw.js (Service Worker 離線快取) 
const CACHE_NAME = 'weya-vote-v1';
const ASSETS_TO_CACHE = [
  './',
  './index.html',
  'https://cdn.tailwindcss.com',
  'https://cdn.jsdelivr.net/npm/chart.js@4.4.1/dist/chart.umd.min.js'
];

self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => {
      return cache.addAll(ASSETS_TO_CACHE);
    })
  );
  self.skipWaiting();
});

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then((keys) => {
      return Promise.all(
        keys.map((key) => {
          if (key !== CACHE_NAME) return caches.delete(key);
        })
      );
    })
  );
  self.clients.claim();
});

self.addEventListener('fetch', (event) => {
  // 不快取 Google Sheet API 或即時 CSV
  if (event.request.url.includes('google.com') || event.request.url.includes('script.google.com')) {
    event.respondWith(fetch(event.request));
    return;
  }
  event.respondWith(
    caches.match(event.request).then((response) => {
      return response || fetch(event.request);
    })
  );
});
